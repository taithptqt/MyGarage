using System.Collections.Generic;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using MyGarage.Models;
using AutoMapper;
using MyGarage.Interfaces;
using MyGarage.DTOs.Requests;
using MyGarage.DTOs.Responses;
using MyGarage.Data;
using System;
using System.Linq;

namespace MyGarage.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ServiceOrderController : ControllerBase
    {
        private readonly IServiceOrderService _serviceOrderService;
        private readonly IMapper _mapper;
        private readonly GarageContext _context;

        public ServiceOrderController(IServiceOrderService serviceOrderService, IMapper mapper, GarageContext context)
        {
            _serviceOrderService = serviceOrderService;
            _mapper = mapper;
            _context = context;
        }

        [HttpGet]
        public async Task<ActionResult<IEnumerable<ServiceOrderResponseDto>>> GetAll()
        {
            var items = await _serviceOrderService.GetAllAsync();
            var dtos = _mapper.Map<IEnumerable<ServiceOrderResponseDto>>(items);
            return Ok(dtos);
        }

        [HttpGet("{id}")]
        public async Task<ActionResult<ServiceOrderResponseDto>> GetById(long id)
        {
            var item = await _serviceOrderService.GetByIdAsync(id);
            if (item == null)
            {
                return NotFound();
            }
            var dto = _mapper.Map<ServiceOrderResponseDto>(item);
            return Ok(dto);
        }

        [HttpPost]
        public async Task<ActionResult<ServiceOrderResponseDto>> Create([FromBody] ServiceOrderCreateDto itemDto)
        {
            var entity = _mapper.Map<ServiceOrder>(itemDto);
            var createdItem = await _serviceOrderService.AddAsync(entity);
            var responseDto = _mapper.Map<ServiceOrderResponseDto>(createdItem);
            return CreatedAtAction(nameof(GetById), new { id = createdItem.Id }, responseDto);
        }

        [HttpPost("workshop-create")]
        public async Task<ActionResult<ServiceOrderResponseDto>> CreateWorkshop([FromBody] WorkshopServiceOrderCreateDto itemDto)
        {
            if (itemDto == null)
            {
                return BadRequest(new { message = "Dữ liệu gửi lên không hợp lệ." });
            }

            await using var tx = await _context.Database.BeginTransactionAsync();
            try
            {
                var executeBranchId = itemDto.ExecuteBranchId ?? itemDto.OriginBranchId;
                if (!executeBranchId.HasValue)
                {
                    return BadRequest(new { message = "Vui lòng chọn chi nhánh làm việc trước khi tạo lệnh." });
                }

                var branchExists = await _context.Branches.AnyAsync(x => x.Id == executeBranchId.Value);
                if (!branchExists)
                {
                    return BadRequest(new { message = "Chi nhánh thực hiện không tồn tại." });
                }

                if (itemDto.ServiceAdvisorId.HasValue)
                {
                    var advisorExists = await _context.Staffs.AnyAsync(x => x.Id == itemDto.ServiceAdvisorId.Value);
                    if (!advisorExists)
                    {
                        return BadRequest(new { message = "Cố vấn dịch vụ không tồn tại." });
                    }
                }

                var finalCustomerId = itemDto.CustomerId;
                var finalVehicleId = itemDto.VehicleId;

                if (itemDto.IsQuickOrder)
                {
                    if (string.IsNullOrWhiteSpace(itemDto.GuestName) || string.IsNullOrWhiteSpace(itemDto.GuestPhone))
                    {
                        return BadRequest(new { message = "Vui lòng nhập đầy đủ họ tên và số điện thoại khách vãng lai." });
                    }

                    var guestPhone = itemDto.GuestPhone.Trim();
                    var customer = await _context.Customers.FirstOrDefaultAsync(x => x.Phone == guestPhone);
                    if (customer == null)
                    {
                        customer = new Customer
                        {
                            FullName = itemDto.GuestName.Trim(),
                            Phone = guestPhone,
                            IsActive = true,
                            Type = "GUEST"
                        };
                        _context.Customers.Add(customer);
                        await _context.SaveChangesAsync();
                    }
                    finalCustomerId = customer.Id;

                    finalVehicleId = await ResolveVehicleIdAsync(
                        allowUseExisting: true,
                        existingVehicleId: null,
                        licensePlate: itemDto.GuestVehiclePlate,
                        modelId: itemDto.GuestVehicleModelId,
                        color: itemDto.GuestVehicleColor,
                        vin: itemDto.GuestVehicleVin,
                        year: itemDto.GuestVehicleYear
                    );
                }
                else
                {
                    if (!finalCustomerId.HasValue)
                    {
                        return BadRequest(new { message = "Vui lòng chọn khách hàng." });
                    }

                    var customerExists = await _context.Customers.AnyAsync(x => x.Id == finalCustomerId.Value);
                    if (!customerExists)
                    {
                        return BadRequest(new { message = "Khách hàng đã chọn không tồn tại." });
                    }

                    if (itemDto.IsNewVehicle)
                    {
                        finalVehicleId = await ResolveVehicleIdAsync(
                            allowUseExisting: false,
                            existingVehicleId: null,
                            licensePlate: itemDto.VehicleLicensePlate,
                            modelId: itemDto.VehicleModelId,
                            color: itemDto.VehicleColor,
                            vin: itemDto.VehicleVin,
                            year: itemDto.VehicleYear
                        );
                    }
                    else
                    {
                        if (!finalVehicleId.HasValue)
                        {
                            return BadRequest(new { message = "Vui lòng chọn xe đã lưu." });
                        }

                        var vehicleExists = await _context.Vehicles.AnyAsync(x => x.Id == finalVehicleId.Value);
                        if (!vehicleExists)
                        {
                            return BadRequest(new { message = "Xe đã chọn không tồn tại." });
                        }
                    }
                }

                if (!finalCustomerId.HasValue || !finalVehicleId.HasValue)
                {
                    return BadRequest(new { message = "Không thể xác định khách hàng hoặc xe." });
                }

                var mappingExists = await _context.CustomerVehicles.AnyAsync(x =>
                    x.CustomerId == finalCustomerId.Value && x.VehicleId == finalVehicleId.Value);
                if (!mappingExists)
                {
                    _context.CustomerVehicles.Add(new CustomerVehicle
                    {
                        CustomerId = finalCustomerId.Value,
                        VehicleId = finalVehicleId.Value,
                        IsCurrentOwner = true,
                        OwnershipStartDate = DateOnly.FromDateTime(DateTime.UtcNow.Date)
                    });
                    await _context.SaveChangesAsync();
                }

                var serviceOrder = new ServiceOrder
                {
                    Code = string.IsNullOrWhiteSpace(itemDto.Code) ? $"SO-{DateTime.UtcNow.Ticks.ToString()[^6..]}" : itemDto.Code.Trim(),
                    CustomerId = finalCustomerId.Value,
                    VehicleId = finalVehicleId.Value,
                    OriginBranchId = itemDto.OriginBranchId ?? executeBranchId.Value,
                    ExecuteBranchId = executeBranchId.Value,
                    OrderCategory = string.IsNullOrWhiteSpace(itemDto.OrderCategory) ? "REPAIR" : itemDto.OrderCategory.Trim(),
                    ServiceAdvisorId = itemDto.ServiceAdvisorId,
                    CustomerNotes = string.IsNullOrWhiteSpace(itemDto.CustomerNotes) ? "Không có ghi chú" : itemDto.CustomerNotes.Trim(),
                    RequiredDepositAmount = itemDto.RequiredDepositAmount ?? 0,
                    OldPartsReturnOption = string.IsNullOrWhiteSpace(itemDto.OldPartsReturnOption) ? "DISPOSE_ALL" : itemDto.OldPartsReturnOption,
                    CheckInTime = itemDto.CheckInTime ?? DateTime.UtcNow,
                    EstimatedHandover = itemDto.EstimatedHandover,
                    PromisedHandoverTime = itemDto.PromisedHandoverTime,
                    Status = string.IsNullOrWhiteSpace(itemDto.Status) ? "NEW" : itemDto.Status.Trim()
                };

                _context.ServiceOrders.Add(serviceOrder);
                await _context.SaveChangesAsync();

                if (itemDto.ServiceIds != null && itemDto.ServiceIds.Count > 0)
                {
                    var distinctServiceIds = itemDto.ServiceIds.Distinct().ToList();
                    var standardServices = await _context.StandardServices
                        .Where(x => distinctServiceIds.Contains(x.Id))
                        .ToListAsync();

                    if (standardServices.Count != distinctServiceIds.Count)
                    {
                        return BadRequest(new { message = "Một hoặc nhiều dịch vụ đã chọn không còn tồn tại." });
                    }

                    var details = standardServices.Select(s =>
                    {
                        var taxRate = s.TaxRate ?? 0;
                        var taxAmount = Math.Round(s.BasePrice * taxRate / 100m, 2);
                        var finalAmount = s.BasePrice + taxAmount;
                        return new ServiceOrderDetail
                        {
                            ServiceOrderId = serviceOrder.Id,
                            StandardServiceId = s.Id,
                            ItemType = "LABOR",
                            ItemName = s.Name,
                            Description = "Dịch vụ khách yêu cầu trước",
                            Quantity = 1,
                            Unit = "Lần",
                            UnitPriceAtTime = s.BasePrice,
                            SubTotal = s.BasePrice,
                            TaxRate = taxRate,
                            TaxAmount = taxAmount,
                            FinalAmount = finalAmount,
                            PayerType = "CUSTOMER",
                            Status = "APPROVED"
                        };
                    });

                    _context.ServiceOrderDetails.AddRange(details);
                    await _context.SaveChangesAsync();
                }

                await tx.CommitAsync();

                var createdItem = await _context.ServiceOrders
                    .Include(x => x.Customer)
                    .Include(x => x.Vehicle).ThenInclude(v => v.Model)
                    .FirstAsync(x => x.Id == serviceOrder.Id);
                var responseDto = _mapper.Map<ServiceOrderResponseDto>(createdItem);
                return CreatedAtAction(nameof(GetById), new { id = createdItem.Id }, responseDto);
            }
            catch (InvalidOperationException ex)
            {
                await tx.RollbackAsync();
                return BadRequest(new { message = ex.Message });
            }
            catch (DbUpdateException ex)
            {
                await tx.RollbackAsync();
                return BadRequest(new { message = ex.InnerException?.Message ?? ex.Message });
            }
            catch (Exception)
            {
                await tx.RollbackAsync();
                return StatusCode(500, new { message = "Có lỗi xảy ra khi tạo lệnh sửa chữa. Không có dữ liệu nào được lưu." });
            }
        }

        private async Task<long> ResolveVehicleIdAsync(
            bool allowUseExisting,
            long? existingVehicleId,
            string? licensePlate,
            long? modelId,
            string? color,
            string? vin,
            int? year)
        {
            if (existingVehicleId.HasValue)
            {
                var exists = await _context.Vehicles.AnyAsync(x => x.Id == existingVehicleId.Value);
                if (!exists)
                {
                    throw new InvalidOperationException("Xe đã chọn không tồn tại.");
                }
                return existingVehicleId.Value;
            }

            if (string.IsNullOrWhiteSpace(licensePlate) || !modelId.HasValue || !year.HasValue)
            {
                throw new InvalidOperationException("Vui lòng nhập đầy đủ thông tin xe mới.");
            }

            var normalizedPlate = NormalizePlate(licensePlate);
            var existingVehicle = await _context.Vehicles.FirstOrDefaultAsync(x => x.LicensePlate == normalizedPlate);
            if (existingVehicle != null)
            {
                if (allowUseExisting)
                {
                    return existingVehicle.Id;
                }
                throw new InvalidOperationException($"Biển số xe {normalizedPlate} đã tồn tại.");
            }

            var modelExists = await _context.VehicleModels.AnyAsync(x => x.Id == modelId.Value);
            if (!modelExists)
            {
                throw new InvalidOperationException("Dòng xe không tồn tại.");
            }

            var vehicle = new Vehicle
            {
                LicensePlate = normalizedPlate,
                ModelId = modelId.Value,
                Color = string.IsNullOrWhiteSpace(color) ? null : color.Trim(),
                Vin = string.IsNullOrWhiteSpace(vin) ? null : vin.Trim(),
                ManufactureYear = year.Value
            };
            _context.Vehicles.Add(vehicle);
            await _context.SaveChangesAsync();
            return vehicle.Id;
        }

        private static string NormalizePlate(string plate)
        {
            return plate.Trim().ToUpperInvariant();
        }

        [HttpPut("{id}")]
        public async Task<ActionResult<ServiceOrderResponseDto>> Update(long id, [FromBody] ServiceOrderUpdateDto itemDto)
        {
            var existingItem = await _serviceOrderService.GetByIdAsync(id);
            if (existingItem == null)
            {
                return NotFound();
            }

            _mapper.Map(itemDto, existingItem);
            var updatedItem = await _serviceOrderService.UpdateAsync(id, existingItem);
            
            if (updatedItem == null)
            {
                return NotFound();
            }
            var responseDto = _mapper.Map<ServiceOrderResponseDto>(updatedItem);
            return Ok(responseDto);
        }

        [HttpDelete("{id}")]
        public async Task<IActionResult> Delete(long id)
        {
            var result = await _serviceOrderService.DeleteAsync(id);
            if (!result)
            {
                return NotFound();
            }
            return NoContent();
        }
    }
}
