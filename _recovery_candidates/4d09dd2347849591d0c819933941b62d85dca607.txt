  return new Intl.NumberFormat("vi-VN").format(n) + " đ";
}

  default: ["Tư vấn theo đúng tình trạng thực tế của xe", "Báo giá minh bạch trước khi thực hiện"],

const MIN_LANDING_CATEGORY_CARDS = 3;
const MAX_LANDING_CATEGORY_CARDS = 6;

const LANDING_HERO_IMAGE =
  "https://images.unsplash.com/photo-1492144534655-ae79c964c9d7?auto=format&fit=crop&w=1800&q=80";
const LANDING_SHOWCASE_IMAGE =
  "https://images.unsplash.com/photo-1619767886558-efdc259cde1a?auto=format&fit=crop&w=1400&q=80";
const LANDING_DETAIL_IMAGE =
  "https://images.unsplash.com/photo-1487754180451-c456f719a1fc?auto=format&fit=crop&w=1400&q=80";
const BOOKING_PREFILL_STORAGE_KEY = "mygarage_booking_prefill";
const BOOKING_PREFILL_TRIGGER_KEY = "mygarage_booking_prefill_trigger";

export default function BookingPage() {
  const navigate = useNavigate();
  const location = useLocation();
  const [step, setStep] = useState(0);

  const [branches, setBranches] = useState([]);
  const [services, setServices] = useState([]);
  const [brands, setBrands] = useState([]);
  const [models, setModels] = useState([]);
  const [products, setProducts] = useState([]);
  const [categories, setCategories] = useState([]);
  const [serviceCategories, setServiceCategories] = useState([]);

  const [selectedBrandId, setSelectedBrandId] = useState("");
  const [selectedModelId, setSelectedModelId] = useState("");

  const [branchId, setBranchId] = useState("");
  const [selectedDate, setSelectedDate] = useState("");
  const [slots, setSlots] = useState([]);
  const [slotsLoading, setSlotsLoading] = useState(false);
  const [selectedSlot, setSelectedSlot] = useState(null);
  const [selectedServiceIds, setSelectedServiceIds] = useState([]);
  const [cart, setCart] = useState({});
  const [activeCatFilter, setActiveCatFilter] = useState("all");
  const [cityFilter, setCityFilter] = useState("");
  const [activeServiceType, setActiveServiceType] = useState("ALL");
  const [licensePlate, setLicensePlate] = useState("");
  const [phone, setPhone] = useState("");
  const [notes, setNotes] = useState("");

  const [guestName, setGuestName] = useState("");
  const [guestPhone, setGuestPhone] = useState("");
  const [guestEmail, setGuestEmail] = useState("");

  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");

  const loggedIn = isAuthenticated();

  useEffect(() => {
    loadReferenceData();
  }, []);

  useEffect(() => {
    const params = new URLSearchParams(location.search);
    const isBookingStartNavigation = params.get("step") === "1" || params.get("start") === "1";
    const isTriggeredFromLanding = sessionStorage.getItem(BOOKING_PREFILL_TRIGGER_KEY) === "1";

    if (!isBookingStartNavigation || !isTriggeredFromLanding) {
      sessionStorage.removeItem(BOOKING_PREFILL_STORAGE_KEY);
      sessionStorage.removeItem(BOOKING_PREFILL_TRIGGER_KEY);
      return;
    }

    try {
      const raw = sessionStorage.getItem(BOOKING_PREFILL_STORAGE_KEY);
      if (!raw) return;
      const prefill = JSON.parse(raw);
      const nextBrandId = prefill?.selectedBrandId ? String(prefill.selectedBrandId) : "";
      const nextModelId = prefill?.selectedModelId ? String(prefill.selectedModelId) : "";
      if (nextBrandId) setSelectedBrandId(nextBrandId);
      if (nextModelId) setSelectedModelId(nextModelId);
    } catch {
      /* ignore */
    } finally {
      sessionStorage.removeItem(BOOKING_PREFILL_STORAGE_KEY);
      sessionStorage.removeItem(BOOKING_PREFILL_TRIGGER_KEY);
    }
  }, [location.search]);

  useEffect(() => {
    const params = new URLSearchParams(location.search);
    const nextStep = Number(params.get("step") || params.get("start") || 0);
    if (!Number.isInteger(nextStep) || nextStep < 1 || nextStep > 5) return;

    setStep(nextStep);
    window.scrollTo({ top: 0, behavior: "auto" });

    params.delete("step");
    params.delete("start");
    const nextSearch = params.toString();
    navigate(
      {
        pathname: location.pathname,
        search: nextSearch ? `?${nextSearch}` : "",
      },
      { replace: true }
    );
  }, [location.pathname, location.search, navigate]);

  async function loadReferenceData() {
    try {
      const [br, svc, vb, vm, prods, cats, svcCats] = await Promise.all([
        getAllBranches(),
        getAllStandardServices(),
        getVehicleBrands(),
        getVehicleModels(),
        getAllProducts(),
        getAllCategories(),
        getAllServiceCategories(),
      ]);
      setBranches(Array.isArray(br) ? br.filter((b) => b.status === "active") : []);
      setServices(Array.isArray(svc) ? svc.filter((s) => s.isActive !== false) : []);
      setBrands(Array.isArray(vb) ? vb.filter((b) => b.isActive !== false) : []);
      setModels(Array.isArray(vm) ? vm : []);
      setProducts(Array.isArray(prods) ? prods.filter((p) => p.isActive !== false && !p.isDeleted) : []);
      setCategories(Array.isArray(cats) ? cats.filter((c) => !c.isDeleted) : []);
      setServiceCategories(Array.isArray(svcCats) ? svcCats : []);
    } catch {
      /* ignore */
    }
  }

  useEffect(() => {
    if (step !== 4) return;
    if (!branchId || !selectedDate) return;

    const requestedServiceIds = selectedServiceIds.length
      ? selectedServiceIds.join(",")
      : null;

    setSelectedSlot(null);
    setSlots([]);
    setSlotsLoading(true);

    getAvailableSlots(Number(branchId), selectedDate, requestedServiceIds)
      .then((data) => setSlots(Array.isArray(data) ? data : []))
      .catch(() => setSlots([]))
      .finally(() => setSlotsLoading(false));
  }, [step, branchId, selectedDate, selectedServiceIds]);

  function toggleService(id) {
    setSelectedServiceIds((prev) =>
      prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]
    );
  }

  function updateCart(productId, delta) {
    setCart((prev) => {
      const current = prev[productId] || 0;
      const next = Math.max(0, current + delta);
      if (next === 0) {
        const copy = { ...prev };
        delete copy[productId];
        return copy;
      }
      return { ...prev, [productId]: next };
    });
  }

  function removeFromCart(productId) {
    setCart((prev) => {
      const copy = { ...prev };
      delete copy[productId];
      return copy;
    });
  }

  function goToStep(s) {
    setStep(s);
    window.scrollTo({ top: 0, behavior: "smooth" });
  }

  const isOtherBrand = selectedBrandId === "other";
  const currentModels = isOtherBrand
    ? OTHER_MODELS
    : models.filter((m) => String(m.brandId) === String(selectedBrandId));

  const selectedBranch = branches.find((b) => String(b.id) === String(branchId));
  const selectedBrand = isOtherBrand
    ? { name: "Khác" }
    : brands.find((b) => String(b.id) === String(selectedBrandId));
  const selectedModel = isOtherBrand
    ? OTHER_MODELS.find((m) => m.id === selectedModelId)
    : models.find((m) => String(m.id) === String(selectedModelId));

  const vehicleName = [selectedBrand?.name, selectedModel?.modelName].filter(Boolean).join(" - ") || "";
  const selectedServices = services.filter((s) => selectedServiceIds.includes(s.id));
  const serviceTotalPrice = selectedServices.reduce((sum, s) => sum + (s.basePrice || 0), 0);
  const totalMinutes = selectedServices.reduce((sum, s) => sum + (s.standardMinutes || 0), 0);

  const serviceCategoryCodeById = useMemo(() => {
    const map = new Map();
    (serviceCategories || []).forEach((c) => {
      if (c?.id != null) map.set(String(c.id), (c.code || "").toUpperCase());
    });
    return map;
  }, [serviceCategories]);

  const isRepairBooking = selectedServices.some((s) =>
    (serviceCategoryCodeById.get(String(s.categoryId)) || "").includes("REPAIR")
  );

  const visibleServices = useMemo(() => {
    if (!activeServiceType || activeServiceType === "ALL") return services;
    return services.filter(
      (s) => serviceCategoryCodeById.get(String(s.categoryId)) === activeServiceType
    );
  }, [activeServiceType, services, serviceCategoryCodeById]);

  const landingCategoryCards = useMemo(() => {
    const activeServiceCategories = (serviceCategories || []).filter(
      (c) => c && c.isDeleted !== true && c.isActive !== false
    );

    const statsByCategory = new Map();
    (services || []).forEach((svc) => {
      const catId = svc?.categoryId != null ? String(svc.categoryId) : "";
      if (!catId) return;

      const current = statsByCategory.get(catId) || {
        serviceCount: 0,
        minutesTotal: 0,
        minPrice: Number.POSITIVE_INFINITY,
      };

      current.serviceCount += 1;

      const minutes = Number(svc.standardMinutes || 0);
      if (minutes > 0) current.minutesTotal += minutes;

      const basePrice = Number(svc.basePrice || 0);
      if (basePrice > 0 && basePrice < current.minPrice) current.minPrice = basePrice;

      statsByCategory.set(catId, current);
    });

    const cards = activeServiceCategories.map((cat, index) => {
      const catId = String(cat.id);
      const stats = statsByCategory.get(catId) || {
        serviceCount: 0,
        minutesTotal: 0,
        minPrice: Number.POSITIVE_INFINITY,
      };
      const code = (cat?.code || "").toUpperCase();
      const preset = LANDING_CATEGORY_PRESETS[code] || {};
      const averageMinutes = stats.serviceCount ? Math.round(stats.minutesTotal / stats.serviceCount) : 0;

      return {
        id: cat?.id || `cat-${catId}`,
        order: cat?.displayOrder ?? 999,
        code,
        title: preset.title || cat?.name || `Danh mục ${index + 1}`,
        description:
          cat?.description ||
          preset.description ||
          "Danh mục được thiết kế theo quy trình chuẩn và tùy theo tình trạng thực tế của xe.",
        icon: preset.icon || "⚙️",
        tone: preset.tone || "default",
        serviceCount: stats.serviceCount,
        averageMinutes,
        minPrice: Number.isFinite(stats.minPrice) ? stats.minPrice : 0,
      };
    });

    const sortedCards = cards.sort(
      (a, b) =>
        Number(b.serviceCount > 0) - Number(a.serviceCount > 0) ||
        a.order - b.order ||
        b.serviceCount - a.serviceCount ||
        a.title.localeCompare(b.title, "vi")
    );

    const usedCodes = new Set(sortedCards.map((c) => c.code).filter(Boolean));
    const presetFallbackCards = Object.entries(LANDING_CATEGORY_PRESETS)
      .filter(([code]) => !usedCodes.has(code))
      .map(([code, preset], index) => ({
        id: `preset-${code}-${index}`,
        order: 999 + index,
        code,
        title: preset.title,
        description: preset.description,
        icon: preset.icon,
        tone: preset.tone,
        serviceCount: 0,
        averageMinutes: 0,
        minPrice: 0,
      }));

    const merged = [...sortedCards];
    let fallbackIndex = 0;
    while (merged.length < MIN_LANDING_CATEGORY_CARDS && fallbackIndex < presetFallbackCards.length) {
      merged.push(presetFallbackCards[fallbackIndex]);
      fallbackIndex += 1;
    }

    if (merged.length === 0) {
      merged.push(
        {
          id: "maint-fallback",
          order: 1,
          code: "MAINTENANCE",
          title: "Bảo dưỡng định kỳ",
          description: "Kiểm tra định kỳ, thay nhớt và vật tư hao mòn theo lịch sử sử dụng xe.",
          icon: "🛠️",
          tone: "maintenance",
          serviceCount: 0,
          averageMinutes: 0,
          minPrice: 0,
        },
        {
          id: "repair-fallback",
          order: 2,
          code: "GENERAL_REPAIR",
          title: "Sửa chữa tổng quát",
          description: "Chẩn đoán lỗi hệ thống và sửa chữa theo quy trình kỹ thuật của gara.",
          icon: "🔧",
          tone: "repair",
          serviceCount: 0,
          averageMinutes: 0,
          minPrice: 0,
        },
        {
          id: "quick-fallback",
          order: 3,
          code: "QUICK_SERVICE",
          title: "Dịch vụ nhanh",
          description: "Các hạng mục bảo trì nhanh, đặt lịch linh hoạt và xử lý trong ngày.",
          icon: "⚡",
          tone: "quick",
          serviceCount: 0,
          averageMinutes: 0,
          minPrice: 0,
        }
      );
    }

    return merged.slice(0, MAX_LANDING_CATEGORY_CARDS).map((card) => {
      const toneKey = card.tone || "default";
      return {
        ...card,
        badge: card.serviceCount > 0 ? "Dịch vụ được khách hàng ưu tiên" : "Tư vấn theo nhu cầu xe",
        highlights: LANDING_CATEGORY_HIGHLIGHTS[toneKey] || LANDING_CATEGORY_HIGHLIGHTS.default,
      };
    });
  }, [serviceCategories, services]);

  const cartItems = useMemo(() => {
    return Object.entries(cart).map(([pid, qty]) => {
      const p = products.find((x) => String(x.id) === String(pid));
      return p ? { ...p, qty } : null;
    }).filter(Boolean);
  }, [cart, products]);

  const productTotalPrice = cartItems.reduce((sum, item) => sum + (item.retailPrice || 0) * item.qty, 0);
  const totalPrice = serviceTotalPrice + productTotalPrice;

  function norm(s) {
    return (s || "")
      .toString()
      .normalize("NFD")
      .replace(/[\u0300-\u036f]/g, "")
      .toLowerCase();
  }

  const accessoryKeywords = useMemo(() => {
    // Map selected services -> keywords. Used to filter by BOTH product name and category name.
    if (!selectedServices.length) return [];

    const kw = new Set();

    for (const s of selectedServices) {
      const catCode = (serviceCategoryCodeById.get(String(s.categoryId)) || "").toUpperCase();
      const name = norm(s.name);
      const svcCode = (s.code || "").toUpperCase();

      const isOilService =
        catCode.includes("MAINT") ||
        svcCode.includes("MAINT") ||
        name.includes("nhot") ||
        name.includes("loc") ||
        name.includes("oil") ||
        name.includes("filter");

      const isWheelService =
        svcCode.includes("TIRE") ||
        catCode.includes("TIRE") ||
        name.includes("lop") ||
        name.includes("banh") ||
        name.includes("phanh") ||
        name.includes("brake");

      // QUICK_SERVICE in your DB can contain both oil and wheel services, so we disambiguate by name.
      if (isOilService) {
        ["dau", "nhot", "oil", "loc", "filter"].forEach((k) => kw.add(k));
      } else if (isWheelService || catCode.includes("QUICK")) {
        ["lop", "tire", "banh", "phanh", "brake"].forEach((k) => kw.add(k));
      }

      if (catCode.includes("REPAIR")) {
        ["phu tung", "phutung", "part", "linh kien"].forEach((k) => kw.add(k));
      }
    }

    return [...kw].map(norm).filter(Boolean);
  }, [selectedServices, serviceCategoryCodeById]);

  useEffect(() => {
    // Avoid showing an empty list due to stale category filter after services change
    setActiveCatFilter("all");
  }, [selectedServiceIds]);

  const filteredProducts = useMemo(() => {
    let list = products;

    // Only show products relevant to selected services
    if (accessoryKeywords.length > 0) {
      const catNameById = new Map(
        (categories || []).map((c) => [String(c.id), norm(c.categoryName)])
      );

      list = list.filter((p) => {
        const pName = norm(p.name);
        const cName = p.categoryId ? (catNameById.get(String(p.categoryId)) || "") : "";
        return accessoryKeywords.some((k) => pName.includes(k) || cName.includes(k));
      });
    } else {
      // If we can't infer anything, don't block the user: show all.
      list = products;
    }

    if (activeCatFilter !== "all") {
      list = list.filter((p) => String(p.categoryId) === String(activeCatFilter));
    }
    return list;
  }, [products, categories, activeCatFilter, accessoryKeywords]);

  const usedCategoryIds = useMemo(() => {
    const ids = new Set();
    filteredProducts.forEach((p) => { if (p.categoryId) ids.add(p.categoryId); });
    return ids;
  }, [filteredProducts]);

  const visibleCategories = categories.filter((c) => usedCategoryIds.has(c.id));

  function extractCity(address) {
    if (!address) return "";
    const parts = address.split(",").map((s) => s.trim());
    return parts[parts.length - 1] || "";
  }

  const branchCities = useMemo(() => {
    const cities = new Set();
    branches.forEach((b) => {
      const city = extractCity(b.address);
      if (city) cities.add(city);
    });
    return [...cities].sort();
  }, [branches]);

  const filteredBranches = useMemo(() => {
    if (!cityFilter) return branches;
    return branches.filter((b) => extractCity(b.address) === cityFilter);
  }, [branches, cityFilter]);

  function selectBranch(id) {
    setBranchId(String(id));
  }

  function getTodayStr() {
    const d = new Date();
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, "0");
    const day = String(d.getDate()).padStart(2, "0");
    return `${y}-${m}-${day}`;
  }

  async function handleSubmit() {
    setError("");
    setSuccess("");
    if (!selectedBrandId) return setError("Vui lòng chọn hãng xe.");
    if (!selectedModelId) return setError("Vui lòng chọn dòng xe.");
    if (!branchId) return setError("Vui lòng chọn chi nhánh.");
    if (!selectedDate) return setError("Vui lòng chọn ngày hẹn.");
    if (!selectedSlot) return setError("Vui lòng chọn giờ hẹn.");
    if (!loggedIn && !guestPhone.trim()) return setError("Vui lòng nhập số điện thoại.");

    setSubmitting(true);
    try {
      const realModelId = isOtherBrand ? null : (selectedModelId ? Number(selectedModelId) : null);
      const startTime = selectedSlot.startTime;
      const endTime = selectedSlot.endTime;

      const payload = {
        branchId: Number(branchId),
        vehicleId: null,
        vehicleModelId: realModelId,
        licensePlate: licensePlate.trim() || null,
        color: null,
        scheduledDate: selectedDate,
        startTime,
        endTime,
        requestedServiceIds: selectedServiceIds.length
          ? selectedServiceIds.join(",")
          : null,
        customerNotes: notes || null,
      };

      if (loggedIn) {
        await bookAppointment(payload);
      } else {
        payload.guestName = guestName.trim() || null;
        payload.guestPhone = guestPhone.trim();
        payload.guestEmail = guestEmail.trim() || null;
        await bookGuestAppointment(payload);
      }

      sessionStorage.removeItem(BOOKING_PREFILL_STORAGE_KEY);
      sessionStorage.removeItem(BOOKING_PREFILL_TRIGGER_KEY);
      setSuccess("Đặt lịch thành công!");
      goToStep(5);
    } catch (err) {
      setError(err.message || "Đặt lịch thất bại.");
    } finally {
      setSubmitting(false);
    }
  }

  /* ============================== STEP 0: Landing ============================== */
  if (step === 0) {
    return (
      <div className="bp-landing">
        <section className="bp-mkt-hero">
          <img className="bp-mkt-hero-image" src={LANDING_HERO_IMAGE} alt="MyGarage workshop" />
          <div className="bp-mkt-hero-overlay" />
          <div className="bp-mkt-hero-content">
            <p className="bp-mkt-kicker">MYGARAGE PREMIUM WORKSHOP</p>
            <h1>Đặt lịch chăm sóc xe theo chuẩn gara chuyên nghiệp</h1>

            {!loggedIn && (
              <div className="booking-card">
                <h3>Thông tin khách hàng</h3>
                <div className="bp-guest-fields">
                  <div className="bp-guest-field">
                    <label>Họ và tên</label>
                    <input type="text" placeholder="Nguyễn Văn A" value={guestName} onChange={(e) => setGuestName(e.target.value)} className="bp-plate-input" />
                  </div>
                  <div className="bp-guest-field">
                    <label>Số điện thoại <span className="bp-required">*</span></label>
                    <input type="tel" placeholder="0901234567" value={guestPhone} onChange={(e) => setGuestPhone(e.target.value)} className="bp-plate-input" />
                  </div>
                  <div className="bp-guest-field">
                    <label>Email</label>
                    <input type="email" placeholder="email@example.com" value={guestEmail} onChange={(e) => setGuestEmail(e.target.value)} className="bp-plate-input" />
                  </div>
                </div>
              </div>
            )}

            {/* License plate + Phone */}
            <div className="booking-card" style={{ marginTop: loggedIn ? 0 : 20 }}>
              <h3>Thông tin xe</h3>
              <div className="bp-guest-fields">
                <div className="bp-guest-field">
                  <label>Hãng xe <span className="bp-required">*</span></label>
                  <select
                    value={selectedBrandId}
                    onChange={(e) => {
                      setSelectedBrandId(e.target.value);
                      setSelectedModelId("");
                    }}
                    className="bp-form-select"
                  >
                    <option value="">Chọn hãng xe</option>
                    {brands.map((b) => (
                      <option key={b.id} value={b.id}>
                        {b.name}
                      </option>
                    ))}
                    <option value="other">Khác</option>
                  </select>
                </div>
                <div className="bp-guest-field">
                  <label>Dòng xe <span className="bp-required">*</span></label>
                  <select
                    value={selectedModelId}
                    onChange={(e) => setSelectedModelId(e.target.value)}
                    disabled={!selectedBrandId}
                    className="bp-form-select"
                  >
                    <option value="">Chọn dòng xe</option>
                    {currentModels.map((m) => (
                      <option key={m.id} value={m.id}>
                        {m.modelName}
                      </option>
                    ))}
                  </select>
                </div>
                <div className="bp-guest-field">
                  <label>Biển số xe (tùy chọn)</label>
                  <input type="text" placeholder="VD: 43A-12345" value={licensePlate} onChange={(e) => setLicensePlate(e.target.value.toUpperCase())} className="bp-plate-input" />
                </div>
                {loggedIn && (
                  <div className="bp-guest-field">
                    <label>Số điện thoại (tùy chọn)</label>
                    <input type="tel" placeholder="0901234567" value={phone} onChange={(e) => setPhone(e.target.value)} className="bp-plate-input" />
                  </div>
                )}
              </div>
            </div>

            {/* Date picker */}
            <div className="booking-card" style={{ marginTop: 20 }}>
              <h3>Ngày và giờ mang xe</h3>
              <div className="bp-datetime-row">
                <div className="bp-datetime-field">
                  <label>Ngày</label>
                  <input
                    type="date"
                    value={selectedDate}
                    onChange={(e) => {
                      setSelectedDate(e.target.value);
                      setSelectedSlot(null);
                    }}
                    min={getTodayStr()}
                    className="bp-date-input"
                  />
                </div>
              </div>

              <div style={{ marginTop: 16 }}>
                <h4 style={{ margin: "0 0 10px", fontSize: 14, color: "#4a5568" }}>
                  Giờ (slot 15 phút)
                </h4>
                <SlotPicker
                  slots={slots}
                  selectedSlot={selectedSlot}
                  onSelect={(s) => setSelectedSlot(s)}
                  loading={slotsLoading}
                  selectedDate={selectedDate ? new Date(selectedDate + "T00:00:00") : null}
                  showEndTime={!isRepairBooking}
                />
                {selectedSlot && !isRepairBooking && (
                  <div style={{ marginTop: 10, fontSize: 13, color: "#422afb", fontWeight: 700 }}>
                    Ước tính kết thúc: {String(selectedSlot.endTime).substring(0, 5)}
                  </div>
                )}
              </div>
            </div>

            {/* Notes */}
            <div className="booking-card" style={{ marginTop: 20 }}>
              <h3>Ghi chú</h3>
              <textarea placeholder="Mô tả triệu chứng hoặc yêu cầu đặc biệt..." value={notes} onChange={(e) => setNotes(e.target.value)} />
            </div>

            <div style={{ marginTop: 20 }}>
              <button className="bp-back-link" onClick={() => goToStep(3)}>← Quay lại chọn cửa hàng</button>
            </div>
          </div>
          {sidebar}
        </div>
      </div>
    );
  }

  /* ============================== STEP 5: Hoàn thành ============================== */
  if (step === 5) {
    return (
      <div className="bp-wizard">
        {progressBar}
        <div className="bp-done">
          <div className="bp-done-icon">✅</div>
          <h2>Đặt lịch thành công!</h2>
          <p>Cảm ơn bạn đã sử dụng dịch vụ của MyGarage.</p>
          <p>Chúng tôi sẽ xác nhận lịch hẹn của bạn trong thời gian sớm nhất.</p>
          <div className="bp-done-actions">
            <button className="bp-done-btn primary" onClick={() => navigate("/my-appointments")}>Xem lịch hẹn của tôi</button>
            <button className="bp-done-btn secondary" onClick={() => { setStep(0); setSelectedServiceIds([]); setCart({}); setBranchId(""); setSelectedDate(""); setSelectedSlot(null); setSlots([]); setLicensePlate(""); setNotes(""); }}>
              Đặt lịch mới
            </button>
          </div>
        </div>
      </div>
    );
  }

  return null;
}
