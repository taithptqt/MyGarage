-- dbo.vw_gl_trial_balance source

ALTER   VIEW dbo.vw_gl_trial_balance AS
          SELECT
              coa.account_code,
              coa.account_name,
              SUM(jel.debit_amount) AS total_debit,
              SUM(jel.credit_amount) AS total_credit,
              SUM(jel.debit_amount - jel.credit_amount) AS balance
          FROM dbo.journal_entry_lines jel
          INNER JOIN dbo.chart_of_accounts coa ON coa.id = jel.account_id
          INNER JOIN dbo.journal_entries je ON je.id = jel.journal_entry_id
          WHERE je.status = 'POSTED'
          GROUP BY coa.account_code, coa.account_name;

ALTER   VIEW dbo.vw_payroll_run_summary AS
          SELECT
              pr.id AS payroll_run_id,
              pr.payroll_period_id,
              pr.branch_id,
              pr.status,
              COUNT(pi.id) AS staff_count,
              SUM(pi.base_salary) AS total_base_salary,
              SUM(pi.allowance_amount) AS total_allowance,
              SUM(pi.overtime_amount) AS total_overtime,
              SUM(pi.commission_amount) AS total_commission,
              SUM(pi.net_pay) AS total_net_pay
          FROM dbo.payroll_runs pr
          LEFT JOIN dbo.payroll_items pi ON pi.payroll_run_id = pr.id
          GROUP BY pr.id, pr.payroll_period_id, pr.branch_id, pr.status;
